   426	    <!-- Firebase Authentication Check -->
   427	    <script>
   428	      const FirebaseAuth = window.FirebaseAuth;
   429	
   430	      function handleAuthentication() {
   431	        const FirebaseAuth = window.FirebaseAuth;
   432	
   433	        if (!FirebaseAuth) {
   434	          console.warn('FirebaseAuth module not yet available, retrying...');
   435	          setTimeout(handleAuthentication, 100);
   436	          return;
   437	        }
   438	
   439	        const userIcon = document.getElementById('user-icon');
   440	
   441	        if (userIcon) {
   442	          try {
   443	            const currentUser = firebase.auth().currentUser;
   444	
   445	            if (currentUser) {
   446	              console.log("Firebase user is logged in:", currentUser.uid);
   447	              userIcon.href = "profile";
   448	              userIcon.classList.add('logged-in');
   449	            } else {
   450	              if (FirebaseAuth.isLoggedIn()) {
   451	                console.log("User is logged in from session");
   452	                userIcon.href = "profile";
   453	                userIcon.classList.add('logged-in');
   454	              } else {
   455	                console.log("No user session found");
   456	                userIcon.href = "login";
   457	                userIcon.classList.remove('logged-in');
   458	              }
   459	            }
   460	          } catch (error) {
   461	            console.error('Error checking authentication state:', error);
   462	            userIcon.href = "login";
   463	            userIcon.classList.remove('logged-in');
   464	          }
   465	        }
   466	      }
   467	
   468	      document.addEventListener('DOMContentLoaded', handleAuthentication);
   469	
   470	      FirebaseAuth.observeAuthState((user) => {
   471	        const userIcon = document.getElementById('user-icon');
   472	
   473	        if (userIcon) {
   474	          if (user) {
   475	            console.log("Auth state: User is signed in:", user.uid);
   476	
   477	            FirebaseAuth.saveSession({
   478	              uid: user.uid,
   479	              email: user.email,
   480	              displayName: user.displayName || 'User',
   481	              photoURL: user.photoURL,
   482	              loggedIn: true
   483	            });
   484	
   485	            userIcon.href = "profile";
   486	            userIcon.classList.add('logged-in');
   487	          } else {
   488	            console.log("Auth state: User is signed out");
   489	            FirebaseAuth.clearSession();
   490	            userIcon.href = "login";
   491	            userIcon.classList.remove('logged-in');
   492	          }
   493	        }
   494	      });
   495	    </script>
