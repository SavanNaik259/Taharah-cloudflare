   470	      const FirebaseAuth = window.FirebaseAuth;
   471	
   472	      function handleAuthentication() {
   473	        const FirebaseAuth = window.FirebaseAuth;
   474	
   475	        if (!FirebaseAuth) {
   476	          console.warn('FirebaseAuth module not yet available, retrying...');
   477	          setTimeout(handleAuthentication, 100);
   478	          return;
   479	        }
   480	
   481	        const userIcon = document.getElementById('user-icon');
   482	
   483	        if (userIcon) {
   484	          try {
   485	            const currentUser = firebase.auth().currentUser;
   486	
   487	            if (currentUser) {
   488	              console.log("Firebase user is logged in:", currentUser.uid);
   489	              userIcon.href = "profile";
   490	              userIcon.classList.add('logged-in');
   491	            } else {
   492	              if (FirebaseAuth.isLoggedIn()) {
   493	                console.log("User is logged in from session");
   494	                userIcon.href = "profile";
   495	                userIcon.classList.add('logged-in');
   496	              } else {
   497	                console.log("No user session found");
   498	                userIcon.href = "login";
   499	                userIcon.classList.remove('logged-in');
   500	              }
   501	            }
   502	          } catch (error) {
   503	            console.error('Error checking authentication state:', error);
   504	            userIcon.href = "login";
   505	            userIcon.classList.remove('logged-in');
   506	          }
   507	        }
   508	      }
   509
